(** ** Derived <<bindd>> Operation *)

Module DerivedOperations.

  #[export] Instance Bindd_Categorical
    (W: Type)
    (T: Type -> Type)
    `{Map_T: Map T}
    `{Join_T: Join T}
    `{Decorate_WT: Decorate W T}: Bindd W T T :=
  fun A B f => join (T := T) ∘ map (F := T) f ∘ dec T.
