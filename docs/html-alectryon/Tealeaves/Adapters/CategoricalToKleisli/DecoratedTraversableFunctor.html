(** ** Derived Operations *)
(**********************************************************************)
Module DerivedOperations.

  #[export] Instance Mapdt_Categorical
    (E: Type)
    (T: Type -> Type)
    `{Map_T: Map T}
    `{Decorate_T: Decorate E T}
    `{Dist_T: ApplicativeDist T}:
  Mapdt E T :=
  fun (G: Type -> Type) `{Map G} `{Pure G} `{Mult G}
      (A B: Type) (f: E * A -> G B) =>
    (dist T G ∘ map f ∘ dec T: T A -> G (T B)).

End DerivedOperations.
