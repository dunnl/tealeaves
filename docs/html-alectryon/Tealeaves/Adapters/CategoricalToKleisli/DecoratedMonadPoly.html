Module DerivedOperations.

  #[export] Instance BinddP_Categorical
    (T: Type -> Type -> Type)
    `{Map2 T}
    `{DecoratePoly T}
    `{forall B, Join (T B)}: BinddP T :=
    fun (B1 B2 V1 V2: Type)
      (ρ: list B1 * B1 -> B2)
      (σ: list B1 * V1 -> T B2 V2)
    => (join ∘ map2 ρ σ ∘ decp (F := T) (B := B1) (V := V1)).

End DerivedOperations.
