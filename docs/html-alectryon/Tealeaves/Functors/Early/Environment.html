    `(f: E * A -> G B) (Γ: env E A): G (env E B) :=
    match Γ with
    | nil => pure (@nil (E * B))

        pure (@List.cons (E * B))

    end.

    `(f: A -> G B) (l: env E A): G (env E B) :=
    match l with
    | nil => pure (@nil (E * B))

    `(f: W * A -> G (env W B)) (Γ: env W A): G (env W B) :=
    match Γ with
    | nil => pure (@nil (W * B))
    | (w, a) :: rest =>
        pure (@List.app (W * B))
          <⋆> map (F := G) (fun x => shift list (w, x)) (f (w, a))
          <⋆> binddt_env G f rest
    end.

    match Γ with
    | nil => @nil (W * B)
    | (w, a) :: rest =>
        shift list (w, f (w, a)) ++ bindd_env f rest

          <⋆> shift list (w, f a)
          <⋆> bind_env f xs
    end.
