(** * The <<hmap>> Operation *)
(**********************************************************************)
(* Fold over a list of A's where each A has the prefix of the list so far *)
Fixpoint hmap {A B: Type}
  (f: list B * A -> B)
  (l: list A): list B :=
  match l with
  | nil => nil
  | cons a rest =>
      f ([], a) :: hmap (f ⦿ [f ([], a)]) rest
  end.
