(** ** Operation <<dist>> *)
(**********************************************************************)
Class ApplicativeDist (F: Type -> Type) :=
  dist: forall (G: Type -> Type) `{Map G} `{Pure G} `{Mult G},
      F ○ G ⇒ G ○ F.
