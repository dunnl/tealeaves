Class PolyDecorateNatural F `{Map2 F} `{DecoratePoly F}: Type :=
  polydecnat: forall (B V B' V': Type) (g: B -> B') (f: V -> V'),
    decp (B := B') (V := V') ∘ map2 g f =
      map2 (map (F := Z) g) (map2 (F := L) g f) ∘ decp (B := B) (V := V).
