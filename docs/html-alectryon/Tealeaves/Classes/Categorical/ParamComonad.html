Section ParameterizedComonad_operations.

    (W: Type -> Type -> Type -> Type).

  Class PExtract :=
    pextract: forall A, W A A ⇒ (fun X => X).

  Class PCojoin :=
    pcojoin: forall A B C, W A C ⇒ W A B ∘ W B C.
