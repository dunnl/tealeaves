(** ** Monad Homomorphisms *)
(**********************************************************************)
Section monad_homomorphism.

  Context
    (T: Type -> Type)
    (U: Type -> Type)
    `{Monad T}
    `{Monad U}.

  Class Monad_Hom (ϕ: T ⇒ U) :=
    { mhom_domain: Monad T;
      mhom_codomain: Monad U;
      mhom_natural: Natural ϕ;
      mhom_ret:
      `(ϕ A ∘ ret T A = ret U A);
      mhom_join:
      `(ϕ A ∘ join T A = join U A ∘ ϕ (U A) ∘ map T (T A) (U A) (ϕ A));
    }.

End monad_homomorphism.
