(** ** Operation <<mapdz>> *)
(******************************************************************************)
Class MapdZ
  (T: Type -> Type) :=
  mapdz:
    forall (A B: Type),
      (Z A -> B) ->
      T A -> T B.

(** ** Kleisli Composition *)
(**********************************************************************)
Definition kc_dz {B1 B2 B3: Type}
  (ρ2: list B2 * B2 -> B3) (* second op to rename binders *)
  (ρ1: list B1 * B1 -> B2) (* first op to rename binders *)
  : list B1 * B1 -> B3 :=
  ρ2 ∘ cobind (W := Z) ρ1.

(** ** Typeclass *)
(**********************************************************************)
Class DecoratedFunctorZ
  (T: Type -> Type) `{MapdZ T} :=
  { kdz_mapdz1:
    forall (A: Type),
      mapdz
        (extract (W := Z))
      = @id (T A);
    kdz_mapdz2:
    forall {B1 B2 B3: Type}
      (ρ1: list B1 * B1 -> B2)
      (ρ2: list B2 * B2 -> B3),
      (mapdz ρ2) ∘ mapdz (T := T) ρ1 =
        mapdz (T := T) (kc_dz ρ2 ρ1)
  }.
